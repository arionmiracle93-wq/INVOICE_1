function setM2CalcUnit(unit){ m2CalcUnit = (unit === 'm') ? 'm' : 'cm'; document.querySelectorAll('.m2-calc-unit-btn').forEach(btn => { btn.classList.toggle('active', btn.dataset.unit === m2CalcUnit); }); document.getElementById('m2CalcUnitW').textContent = m2CalcUnit; document.getElementById('m2CalcUnitH').textContent = m2CalcUnit; updateM2CalcResult(); } function parseCalcNum(v){ const s = String(v).trim(); if(!s) return 0; const cleaned = s.replace(/\s/g, ''); const num = parseFloat(cleaned.replace(',', '.')); return isNaN(num) ? 0 : num; } function updateM2CalcResult(){ const w = parseCalcNum(document.getElementById('m2CalcW').value); const h = parseCalcNum(document.getElementById('m2CalcH').value); const factor = m2Factor(); const m2 = Math.round((w / factor) * (h / factor) * 100) / 100; document.getElementById('m2CalcResult').textContent = m2 > 0 ? m2.toLocaleString('id-ID', {minimumFractionDigits: 2, maximumFractionDigits: 2}) + ' m²' : '0,00 m²'; } function positionPopoverNear(trigger){ const popover = document.getElementById('m2CalcPopover'); const rect = trigger.getBoundingClientRect(); const popH = 210; // approx popover height const popW = 270; let left = Math.max(8, Math.min(rect.left, window.innerWidth - popW - 8)); let top = rect.bottom + 8; // If overflows bottom, show above the trigger if(top + popH > window.innerHeight - 8){ top = Math.max(8, rect.top - popH - 8); } popover.style.left = left + 'px'; popover.style.top = top + 'px'; } function openM2Calc(e){ e && e.stopPropagation(); // Use the tracked last focused m² input, fallback to active element, then last row m2CalcTarget = lastM2Input; if(!m2CalcTarget || !m2CalcTarget.closest('tr')){ const focused = document.activeElement; if(focused && focused.classList.contains('m2')){ m2CalcTarget = focused; } else { const lastRow = document.querySelector('#rows tr:last-child'); m2CalcTarget = lastRow ? lastRow.querySelector('.m2') : null; } } document.getElementById('m2CalcW').value = ''; document.getElementById('m2CalcH').value = ''; setM2CalcUnit('cm'); const trigger = e && e.currentTarget ? e.currentTarget : document.querySelector('.add-row-btn'); if(trigger) positionPopoverNear(trigger); document.getElementById('m2CalcPopover').classList.add('open'); document.getElementById('m2CalcW').focus(); } function closeM2Calc(){ document.getElementById('m2CalcPopover').classList.remove('open'); m2CalcTarget = null; } function applyM2Calc(){ const w = parseCalcNum(document.getElementById('m2CalcW').value); const h = parseCalcNum(document.getElementById('m2CalcH').value); const factor = m2Factor(); const m2 = Math.round((w / factor) * (h / factor) * 100) / 100; if(m2 <= 0 || !m2CalcTarget){ closeM2Calc(); return; } // Write the value directly, bypassing formatM2Input to avoid re-parsing issues const raw = String(m2).replace('.', ','); // Indonesian decimal comma m2CalcTarget.value = raw; // Manually recalc the row const tr = m2CalcTarget.closest('tr'); if(tr) recalcRow(tr); closeM2Calc(); scheduleAutosave(); } // Close calculator when clicking outside document.addEventListener('mousedown', function(e){ if(!e.target.closest('.m2-calc-popover') && !e.target.closest('.add-row-btn')){ closeM2Calc(); } }); document.addEventListener('keydown', function(e){ if(e.key === 'Escape' && document.getElementById('m2CalcPopover').classList.contains('open')){ closeM2Calc(); } }); // Live update result as user types ['m2CalcW', 'm2CalcH'].forEach(id => { document.getElementById(id).addEventListener('input', updateM2CalcResult); document.getElementById(id).addEventListener('keydown', function(e){ if(e.key === 'Enter'){ e.preventDefault(); applyM2Calc(); } }); });
